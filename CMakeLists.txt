# Parity with build.sh, for MSVC and IDE builds.
cmake_minimum_required(VERSION 3.16)
project(aarch64_emu_cpp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB SRC ${CMAKE_SOURCE_DIR}/src/*.cpp)
add_executable(aarch64emu ${SRC})
target_include_directories(aarch64emu PRIVATE src)
if(MSVC)
  target_compile_options(aarch64emu PRIVATE /utf-8 /EHsc /W3)
  target_compile_definitions(aarch64emu PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(aarch64emu PRIVATE -Wall -Wextra)
endif()
